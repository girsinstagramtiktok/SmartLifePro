package com.smartlife.pro;

import android.content.ComponentName;
import android.content.Context;
import android.content.SharedPreferences;
import android.os.Build;
import android.provider.Settings;
import android.service.notification.NotificationListenerService;
import androidx.annotation.NonNull;
import androidx.work.ExistingPeriodicWorkPolicy;
import androidx.work.PeriodicWorkRequest;
import androidx.work.Worker;
import androidx.work.WorkerParameters;
import androidx.work.WorkManager;
import java.util.concurrent.TimeUnit;

/** Recuperación conservadora del listener: nunca cambia permisos ni lee datos bancarios. */
public class SmartyBankHealthWorker extends Worker {
  private static final String WORK_ID = "smarty_bank_listener_health_v1";
  private static final long MIN_RETRY_MS = 60_000L;

  public SmartyBankHealthWorker(@NonNull Context c, @NonNull WorkerParameters p) { super(c,p); }

  public static boolean hasAccess(Context context) {
    String raw = Settings.Secure.getString(context.getContentResolver(), "enabled_notification_listeners");
    if (raw == null) return false;
    ComponentName target = new ComponentName(context, SmartyNotificationService.class);
    for (String entry : raw.split(":")) {
      ComponentName parsed = ComponentName.unflattenFromString(entry.trim());
      if (target.equals(parsed)) return true;
    }
    return false;
  }

  public static void schedule(Context context) {
    try {
      PeriodicWorkRequest job = new PeriodicWorkRequest.Builder(
        SmartyBankHealthWorker.class, 15, TimeUnit.MINUTES).build();
      WorkManager.getInstance(context).enqueueUniquePeriodicWork(
        WORK_ID, ExistingPeriodicWorkPolicy.KEEP, job);
    } catch (Exception ex) { logError(context, "SCHEDULE", ex); }
  }

  public static String requestReconnect(Context context, boolean force) {
    if (!hasAccess(context)) return "permission_missing";
    if (SmartyNotificationService.connected) return "connected";
    if (Build.VERSION.SDK_INT < 24) return "unsupported_android";
    SharedPreferences d = context.getSharedPreferences("smarty_bank_diag", Context.MODE_PRIVATE);
    long now = System.currentTimeMillis();
    if (!force && now - d.getLong("lastRebindAt", 0) < MIN_RETRY_MS) return "throttled";
    try {
      NotificationListenerService.requestRebind(new ComponentName(context, SmartyNotificationService.class));
      d.edit().putLong("lastRebindAt", now)
        .putInt("rebindCount",d.getInt("rebindCount",0)+1).apply();
      return "requested";
    } catch (Exception ex) {
      logError(context,"REBIND",ex);
      return "error:" + ex.getClass().getSimpleName();
    }
  }

  private static void logError(Context c, String step, Exception e) {
    c.getSharedPreferences("smarty_bank_diag", Context.MODE_PRIVATE).edit()
      .putString("lastError", step + ": " + e.getClass().getSimpleName()).apply();
  }

  @NonNull @Override public Result doWork() {
    // Solo comprobar permisos y pedir al sistema que restablezca la conexión.
    requestReconnect(getApplicationContext(), false);
    return Result.success();
  }
}
