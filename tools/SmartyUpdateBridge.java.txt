package com.smartlife.pro;

import android.content.ContentResolver;
import android.content.ContentValues;
import android.content.Context;
import android.content.SharedPreferences;
import android.content.pm.PackageInfo;
import android.content.pm.PackageManager;
import android.content.pm.Signature;
import android.net.Uri;
import android.os.Build;
import android.os.Environment;
import android.provider.MediaStore;
import android.webkit.JavascriptInterface;
import java.io.File;
import java.io.FileInputStream;
import java.io.FileOutputStream;
import java.io.InputStream;
import java.io.OutputStream;
import java.net.HttpURLConnection;
import java.net.URL;
import java.security.MessageDigest;
import java.util.HashSet;
import java.util.Locale;
import java.util.Set;
import java.util.concurrent.ExecutorService;
import java.util.concurrent.Executors;
import org.json.JSONObject;

/** Descarga segura de APK a Descargas/SmartLifePro. NO inicia el instalador. */
public final class SmartyUpdateBridge {
    private final MainActivity host;
    private final ExecutorService worker=Executors.newSingleThreadExecutor();
    private volatile boolean busy=false;
    private static final long MAX_APK=200L*1024L*1024L;
    SmartyUpdateBridge(MainActivity h){host=h;}
    private SharedPreferences prefs(){return host.getSharedPreferences("smarty_update_downloads",Context.MODE_PRIVATE);}
    private void emit(String kind,String msg){host.runOnUiThread(()->{
        try{host.getBridge().getWebView().evaluateJavascript("window.SmartyAppUpdater?.nativeEvent("+JSONObject.quote(kind)+","+JSONObject.quote(msg)+")",null);}catch(Exception ignored){}
    });}
    private static long version(PackageInfo p){return Build.VERSION.SDK_INT>=28?p.getLongVersionCode():p.versionCode;}
    @JavascriptInterface public long getInstalledVersionCode(){
        try{return version(host.getPackageManager().getPackageInfo(host.getPackageName(),0));}catch(Exception e){return 0;}
    }
    private static String hex(byte[] b){StringBuilder out=new StringBuilder(b.length*2);for(byte a:b)out.append(String.format(Locale.ROOT,"%02x",a&255));return out.toString();}
    private static Set<String> signatures(PackageInfo p)throws Exception{
        Signature[] signatures=Build.VERSION.SDK_INT>=28?p.signingInfo.getApkContentsSigners():p.signatures;
        Set<String> out=new HashSet<>();if(signatures!=null)for(Signature s:signatures)out.add(hex(MessageDigest.getInstance("SHA-256").digest(s.toByteArray())));
        return out;
    }
    @SuppressWarnings("deprecation") private void verify(File file,long expected)throws Exception{
        PackageManager pm=host.getPackageManager();int flags=Build.VERSION.SDK_INT>=28?PackageManager.GET_SIGNING_CERTIFICATES:PackageManager.GET_SIGNATURES;
        PackageInfo downloaded=pm.getPackageArchiveInfo(file.getAbsolutePath(),flags),installed=pm.getPackageInfo(host.getPackageName(),flags);
        if(downloaded==null||!host.getPackageName().equals(downloaded.packageName))throw new SecurityException("APK de otra aplicación");
        if(version(downloaded)!=expected||version(downloaded)<=version(installed))throw new SecurityException("Número de versión incompatible");
        if(signatures(installed).isEmpty()||!signatures(installed).equals(signatures(downloaded)))throw new SecurityException("Firma distinta al APK instalado");
    }
    private static boolean allowedHost(String h){return "github.com".equals(h)||"release-assets.githubusercontent.com".equals(h)||"objects.githubusercontent.com".equals(h);}
    private HttpURLConnection connection(String link)throws Exception{
        for(int i=0;i<7;i++){
            URL url=new URL(link);if(!"https".equalsIgnoreCase(url.getProtocol())||!allowedHost(url.getHost()))throw new SecurityException("Servidor de descarga no autorizado");
            HttpURLConnection c=(HttpURLConnection)url.openConnection();c.setInstanceFollowRedirects(false);c.setConnectTimeout(15000);c.setReadTimeout(30000);c.setRequestProperty("User-Agent","SmartLifePro-Updater");
            int status=c.getResponseCode();
            if(status==301||status==302||status==303||status==307||status==308){String redirect=c.getHeaderField("Location");c.disconnect();if(redirect==null)throw new Exception("Redirección vacía");link=new URL(url,redirect).toExternalForm();continue;}
            if(status!=200){c.disconnect();throw new Exception("HTTP "+status);}return c;
        }
        throw new SecurityException("Demasiadas redirecciones");
    }
    private String publish(File source,long version)throws Exception{
        if(Build.VERSION.SDK_INT<29)throw new UnsupportedOperationException("Guardar en Descargas requiere Android 10 o superior en esta versión");
        ContentResolver cr=host.getContentResolver();
        String name="SmartLifePro_v"+version+".apk";
        ContentValues values=new ContentValues();values.put(MediaStore.Downloads.DISPLAY_NAME,name);
        values.put(MediaStore.Downloads.MIME_TYPE,"application/vnd.android.package-archive");
        values.put(MediaStore.Downloads.RELATIVE_PATH,Environment.DIRECTORY_DOWNLOADS+"/SmartLifePro");
        values.put(MediaStore.Downloads.IS_PENDING,1);
        Uri dest=cr.insert(MediaStore.Downloads.EXTERNAL_CONTENT_URI,values);
        if(dest==null)throw new Exception("No se pudo crear APK en Descargas");
        boolean ok=false;
        try(InputStream in=new FileInputStream(source);OutputStream out=cr.openOutputStream(dest)){
            if(out==null)throw new Exception("No se pudo escribir en Descargas");
            byte[] b=new byte[32768];int n;while((n=in.read(b))!=-1)out.write(b,0,n);out.flush();ok=true;
        }finally{if(!ok)cr.delete(dest,null,null);}
        ContentValues done=new ContentValues();done.put(MediaStore.Downloads.IS_PENDING,0);cr.update(dest,done,null,null);
        String path="Descargas/SmartLifePro/"+name;
        prefs().edit().putString("file",path).putLong("version",version).apply();
        return path;
    }
    @JavascriptInterface public synchronized void downloadAndSave(String link,String sha256,String versionString){
        if(busy){emit("error","Otra descarga sigue activa");return;}
        if(sha256==null||!sha256.matches("(?i)[0-9a-f]{64}")){emit("error","SHA-256 inválido");return;}
        final long expected;
        try{expected=Long.parseLong(versionString);if(expected<=getInstalledVersionCode())throw new Exception();}
        catch(Exception e){emit("error","Versión no superior a la instalada");return;}
        if(link==null||!link.matches("https://github\\.com/[A-Za-z0-9_.-]+/[A-Za-z0-9_.-]+/releases/download/[^\\s]+/SmartLifePro\\.apk")){
            emit("error","Dirección de descarga inválida");return;
        }
        busy=true;
        worker.execute(()->{
            File tmp=new File(host.getCacheDir(),"smarty-download-check.apk");
            try{
                MessageDigest md=MessageDigest.getInstance("SHA-256");HttpURLConnection c=connection(link);
                long len=c.getContentLengthLong();if(len>MAX_APK)throw new SecurityException("APK demasiado grande");
                long total=0;int prev=-1;
                try(InputStream in=c.getInputStream();FileOutputStream out=new FileOutputStream(tmp)){
                    byte[] buf=new byte[32768];int n;while((n=in.read(buf))!=-1){total+=n;if(total>MAX_APK)throw new SecurityException("APK supera límite de tamaño");md.update(buf,0,n);out.write(buf,0,n);
                        if(len>0){int pct=(int)(total*100/len);if(pct>=prev+5||pct==100){prev=pct;emit("progress",Integer.toString(pct));}}
                    }
                    out.getFD().sync();
                }finally{c.disconnect();}
                if(!hex(md.digest()).equalsIgnoreCase(sha256))throw new SecurityException("SHA-256 del APK no coincide");
                verify(tmp,expected);
                emit("saved",publish(tmp,expected));
            }catch(Exception e){emit("error",e.getClass().getSimpleName()+": "+e.getMessage());}
            finally{tmp.delete();busy=false;}
        });
    }
    @JavascriptInterface public String getDownloadDiagnostics(){
        JSONObject json=new JSONObject();try{json.put("androidSdk",Build.VERSION.SDK_INT);json.put("lastDownload",prefs().getString("file",""));json.put("lastVersion",prefs().getLong("version",0));json.put("installedVersion",getInstalledVersionCode());json.put("storage","MediaStore.Downloads");}catch(Exception ignored){}return json.toString();
    }
}
