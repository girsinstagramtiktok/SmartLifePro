package com.smartlife.pro;

import android.app.Activity;
import android.content.ActivityNotFoundException;
import android.content.Intent;
import android.content.ClipData;
import android.content.pm.PackageInfo;
import android.content.pm.PackageManager;
import android.content.pm.Signature;
import android.net.Uri;
import android.os.Build;
import android.provider.Settings;
import android.webkit.JavascriptInterface;
import androidx.core.content.FileProvider;
import java.io.File;
import java.io.FileOutputStream;
import java.io.InputStream;
import java.net.HttpURLConnection;
import java.net.URL;
import java.security.MessageDigest;
import java.util.Arrays;
import java.util.HashSet;
import java.util.Locale;
import java.util.Set;
import java.util.concurrent.ExecutorService;
import java.util.concurrent.Executors;
import org.json.JSONObject;

/** APK download and installation bridge. Never exposes signing keys to JavaScript. */
public final class SmartyUpdateBridge {
    private final MainActivity host;
    private final ExecutorService queue=Executors.newSingleThreadExecutor();
    private volatile boolean busy=false;
    private volatile String lastInstallerAction="sin intento";
    private volatile String lastInstallerError="";
    private static final long MAX_APK=200L*1024L*1024L;
    public SmartyUpdateBridge(MainActivity activity){this.host=activity;}

    private void emit(String event,String message){
        host.runOnUiThread(() -> {
            try {
                String script="window.SmartyAppUpdater?.nativeEvent("+JSONObject.quote(event)+","+JSONObject.quote(message)+")";
                host.getBridge().getWebView().evaluateJavascript(script,null);
            }catch(Exception ignored){}
        });
    }
    private File apk(){return new File(new File(host.getCacheDir(),"updates"),"SmartLifePro.apk");}
    private static long version(PackageInfo info){return Build.VERSION.SDK_INT>=28?info.getLongVersionCode():info.versionCode;}
    @JavascriptInterface public long getInstalledVersionCode(){
        try {return version(host.getPackageManager().getPackageInfo(host.getPackageName(),0));}
        catch(Exception ignored){return 0;}
    }
    private static Set<String> certificates(PackageInfo info) throws Exception {
        Signature[] sigs=Build.VERSION.SDK_INT>=28?info.signingInfo.getApkContentsSigners():info.signatures;
        Set<String> out=new HashSet<>();
        if(sigs==null)return out;
        for(Signature s:sigs)out.add(hex(MessageDigest.getInstance("SHA-256").digest(s.toByteArray())));
        return out;
    }
    private static String hex(byte[] arr){StringBuilder s=new StringBuilder(arr.length*2);for(byte b:arr)s.append(String.format(Locale.ROOT,"%02x",b&255));return s.toString();}
    @SuppressWarnings("deprecation")
    private void verify(File file,long expected) throws Exception {
        PackageManager pm=host.getPackageManager();
        int flags=Build.VERSION.SDK_INT>=28?PackageManager.GET_SIGNING_CERTIFICATES:PackageManager.GET_SIGNATURES;
        PackageInfo incoming=pm.getPackageArchiveInfo(file.getAbsolutePath(),flags);
        PackageInfo installed=pm.getPackageInfo(host.getPackageName(),flags);
        if(incoming==null||!host.getPackageName().equals(incoming.packageName))throw new SecurityException("APK de otra aplicación");
        if(version(incoming)<=version(installed)||version(incoming)!=expected)throw new SecurityException("Versión APK incompatible");
        if(!certificates(incoming).equals(certificates(installed))||certificates(installed).isEmpty())throw new SecurityException("Firma diferente. No se pueden conservar los datos con este APK");
    }
    private static boolean allowedHost(String name){
        return "github.com".equals(name)||"release-assets.githubusercontent.com".equals(name)||"objects.githubusercontent.com".equals(name);
    }
    private HttpURLConnection connection(String link) throws Exception {
        for(int attempt=0;attempt<7;attempt++){
            URL u=new URL(link);
            if(!"https".equalsIgnoreCase(u.getProtocol())||!allowedHost(u.getHost()))throw new SecurityException("Servidor de actualización no autorizado");
            HttpURLConnection c=(HttpURLConnection)u.openConnection();
            c.setInstanceFollowRedirects(false);c.setConnectTimeout(15000);c.setReadTimeout(30000);
            c.setRequestProperty("User-Agent","SmartLifePro-Updater");
            int code=c.getResponseCode();
            if(code==301||code==302||code==303||code==307||code==308){
                String next=c.getHeaderField("Location");c.disconnect();
                if(next==null)throw new SecurityException("Redirección inválida");
                link=new URL(u,next).toExternalForm();continue;
            }
            if(code!=200){c.disconnect();throw new IllegalStateException("HTTP "+code);}
            return c;
        }
        throw new SecurityException("Demasiadas redirecciones");
    }
    @JavascriptInterface public synchronized void downloadAndInstall(String url,String sha256,String versionCodeString){
        if(busy){emit("error","Ya hay una descarga en curso");return;}
        if(sha256==null||!sha256.matches("(?i)[0-9a-f]{64}")){emit("error","SHA-256 inválido");return;}
        final long expected;
        try{expected=Long.parseLong(versionCodeString);if(expected<=getInstalledVersionCode())throw new Exception();}
        catch(Exception e){emit("error","Número de versión inválido");return;}
        if(url==null||!url.matches("https://github\\.com/[A-Za-z0-9_.-]+/[A-Za-z0-9_.-]+/releases/download/[^\\s]+/SmartLifePro\\.apk")){
            emit("error","URL APK inválida");return;
        }
        busy=true;
        queue.execute(() -> {
            File output=apk();
            try {
                File dir=output.getParentFile();if(!dir.exists()&&!dir.mkdirs())throw new Exception("No hay espacio para guardar APK");
                File tmp=new File(dir,"download-temp.apk");
                MessageDigest md=MessageDigest.getInstance("SHA-256");
                HttpURLConnection c=connection(url);
                long length=c.getContentLengthLong();
                if(length>MAX_APK)throw new SecurityException("APK demasiado grande");
                long written=0;int oldProgress=-1;
                try(InputStream in=c.getInputStream();FileOutputStream out=new FileOutputStream(tmp)){
                    byte[] bytes=new byte[32768];int n;
                    while((n=in.read(bytes))!=-1){
                        written+=n;if(written>MAX_APK)throw new SecurityException("APK supera el límite permitido");
                        md.update(bytes,0,n);out.write(bytes,0,n);
                        if(length>0){int pct=(int)(written*100/length);if(pct>=oldProgress+5||pct==100){oldProgress=pct;emit("progress",Integer.toString(pct));}}
                    }
                    out.getFD().sync();
                }finally{c.disconnect();}
                if(!hex(md.digest()).equalsIgnoreCase(sha256))throw new SecurityException("El SHA-256 no coincide");
                verify(tmp,expected);
                if(output.exists()&&!output.delete())throw new Exception("No se pudo sustituir APK anterior");
                if(!tmp.renameTo(output))throw new Exception("Error al preparar APK");
                emit("ready","verified");
                install(output,expected);
            }catch(Exception e){emit("error",e.getClass().getSimpleName()+": "+String.valueOf(e.getMessage()));}
            finally{busy=false;}
        });
    }
    /** Diagnóstico real del APK + FileProvider, no solo del permiso de instalación. */
    @JavascriptInterface public String getInstallDiagnostics(){
        JSONObject result=new JSONObject();
        try {
            File file=apk();
            PackageManager pm=host.getPackageManager();
            result.put("androidSdk",Build.VERSION.SDK_INT);
            result.put("apkPresent",file.isFile());
            result.put("apkSizeBytes",file.isFile()?file.length():0);
            result.put("packageName",host.getPackageName());
            result.put("canInstallUnknownApps",Build.VERSION.SDK_INT<26||pm.canRequestPackageInstalls());
            result.put("lastInstallerAction",lastInstallerAction);
            result.put("lastInstallerError",lastInstallerError);
            // La visibilidad de paquetes de Android 11+ hace que resolveActivity
            // pueda devolver null incluso si el instalador sí está disponible.
            result.put("installerMethod","ACTION_VIEW con MIME de APK (FileProvider)");
            if(file.isFile()){
                int flags=Build.VERSION.SDK_INT>=28?PackageManager.GET_SIGNING_CERTIFICATES:PackageManager.GET_SIGNATURES;
                PackageInfo incoming=pm.getPackageArchiveInfo(file.getAbsolutePath(),flags);
                PackageInfo installed=pm.getPackageInfo(host.getPackageName(),flags);
                result.put("apkParseable",incoming!=null);
                result.put("installedVersionCode",version(installed));
                if(incoming!=null){
                    result.put("apkVersionCode",version(incoming));
                    result.put("apkVersionIsNewer",version(incoming)>version(installed));
                    result.put("apkPackageMatches",host.getPackageName().equals(incoming.packageName));
                    try{result.put("apkSignatureMatches",!certificates(installed).isEmpty()&&certificates(incoming).equals(certificates(installed)));}
                    catch(Exception e){result.put("apkSignatureError",String.valueOf(e.getMessage()));}
                }
                try{
                    Uri uri=FileProvider.getUriForFile(host,host.getPackageName()+".smartyupdates",file);
                    result.put("fileProviderUriValid","content".equals(uri.getScheme()));
                }catch(Exception e){result.put("fileProviderError",e.getClass().getSimpleName()+": "+String.valueOf(e.getMessage()));}
            }
        }catch(Exception e){
            try{result.put("diagnosticError",e.getClass().getSimpleName()+": "+String.valueOf(e.getMessage()));}
            catch(Exception ignored){}
        }
        return result.toString();
    }

    /** Abre primero el manejador MIME real de archivos APK en Android/Samsung. */
    private void install(File file,long expected){
        host.runOnUiThread(() -> {
            try {
                verify(file,expected);
                if(Build.VERSION.SDK_INT>=26&&!host.getPackageManager().canRequestPackageInstalls()){
                    lastInstallerAction="permission";
                    lastInstallerError="Permiso de instalación pendiente";
                    emit("permission","SmartLifePro necesita permiso para instalar APK desde esta fuente.");
                    Intent settings=new Intent(Settings.ACTION_MANAGE_UNKNOWN_APP_SOURCES,Uri.parse("package:"+host.getPackageName()));
                    host.startActivity(settings);
                    return;
                }
                Uri uri=FileProvider.getUriForFile(host,host.getPackageName()+".smartyupdates",file);
                // ACTION_INSTALL_PACKAGE es obsoleto desde API 29 y puede no abrir
                // una ventana en algunas ROM. ACTION_VIEW con MIME es el primer intento.
                Intent view=new Intent(Intent.ACTION_VIEW);
                view.setDataAndType(uri,"application/vnd.android.package-archive");
                view.addFlags(Intent.FLAG_GRANT_READ_URI_PERMISSION);
                view.setClipData(ClipData.newRawUri("SmartLifePro.apk",uri));
                try {
                    lastInstallerAction="ACTION_VIEW";
                    lastInstallerError="";
                    host.startActivity(view);
                }catch(ActivityNotFoundException missing){
                    lastInstallerAction="ACTION_INSTALL_PACKAGE (fallback)";
                    Intent fallback=new Intent(Intent.ACTION_INSTALL_PACKAGE);
                    fallback.setData(uri);
                    fallback.addFlags(Intent.FLAG_GRANT_READ_URI_PERMISSION);
                    fallback.setClipData(ClipData.newRawUri("SmartLifePro.apk",uri));
                    host.startActivity(fallback);
                }
                emit("installing","Solicitud enviada mediante "+lastInstallerAction+". Si no aparece el instalador, revisa el diagnóstico: versión, firma, FileProvider y Bloqueador automático de Samsung.");
            }catch(ActivityNotFoundException e){
                lastInstallerError="No se encuentra instalador Android: "+String.valueOf(e.getMessage());
                emit("error",lastInstallerError);
            }catch(SecurityException e){
                lastInstallerError="Android bloqueó la instalación: "+String.valueOf(e.getMessage());
                emit("error",lastInstallerError);
            }catch(Exception e){
                lastInstallerError="No se abrió el instalador: "+e.getClass().getSimpleName()+": "+String.valueOf(e.getMessage());
                emit("error",lastInstallerError);
            }
        });
    }
    /** Responde inmediatamente a JavaScript sin depender de eventos asíncronos. */
    @JavascriptInterface public String installDownloadedApk(){
        File file=apk();
        if(!file.isFile()){
            emit("error","El APK ya no está en el almacenamiento temporal. Descárgalo nuevamente.");
            return "APK no encontrado. Descarga la actualización de nuevo.";
        }
        if(busy)return "Finalizando descarga, vuelve a intentarlo.";
        queue.execute(() -> {
            try{
                PackageInfo p=host.getPackageManager().getPackageArchiveInfo(file.getAbsolutePath(),0);
                if(p==null)throw new Exception("Archivo APK inválido");
                install(file,version(p));
            }catch(Exception e){emit("error","Error al preparar instalador: "+String.valueOf(e.getMessage()));}
        });
        return "Solicitando la instalación a Android…";
    }
}
